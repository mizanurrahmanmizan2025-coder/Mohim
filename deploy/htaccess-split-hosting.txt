# ─────────────────────────────────────────────────────────────────────────────
#  ONLY needed for the "split" setup (Option B in DEPLOY.md):
#  React build uploaded to public_html as static files, API running elsewhere
#  (e.g. a Node.js app on api.yourdomain.com).
#
#  With the recommended single Node.js app (Option A), Hostinger generates the
#  .htaccess for you — do not upload this file in that case.
#
#  Rename to ".htaccess" and place it in public_html next to index.html.
# ─────────────────────────────────────────────────────────────────────────────

Options -MultiViews -Indexes
RewriteEngine On

# Force HTTPS
RewriteCond %{HTTPS} off
RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

# Never let static hosting answer /api — the frontend calls the API domain directly
# (build with VITE_API_URL=https://api.yourdomain.com)
RewriteRule ^api/ - [R=404,L]

# Serve real files/folders as-is
RewriteCond %{REQUEST_FILENAME} -f [OR]
RewriteCond %{REQUEST_FILENAME} -d
RewriteRule ^ - [L]

# SPA fallback: /c/123, /admin, /share/abc → index.html (React Router handles it)
RewriteRule ^ index.html [L]

# Caching: hashed assets forever, index.html never
<IfModule mod_headers.c>
  <FilesMatch "\.(js|css|woff2?|svg|png|jpg|webp)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>
  <FilesMatch "index\.html$">
    Header set Cache-Control "no-cache"
  </FilesMatch>
  Header always set X-Content-Type-Options "nosniff"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
</IfModule>

<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css application/javascript application/json image/svg+xml
</IfModule>
